```python
from PIL import Image
from rembg import remove

# Load files
subject = Image.open("IMG_5290-EDIT.jpg")
bg = Image.open("voronoi-negotico-4x5.jpg")

# Extract person and resize background to match portrait dimensions
subject_no_bg = remove(subject)
bg_resized = bg.resize(subject.size)

# Composite subject over the Voronoi pattern
bg_resized.paste(subject_no_bg, (0, 0), subject_no_bg)
bg_resized.save("portrait_voronoi.jpg")
```
